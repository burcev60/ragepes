Какой из следующих фрагментов кода корректно реализует декоратор с параметрами, который добавляет префикс к результату декорируемой функции?

- A.

    def with_prefix(prefix):
        def decorator(func):
            def wrapper(*args, **kwargs):
                result = func(*args, **kwargs)
                return prefix + str(result)
            return wrapper
        return decorator

   

- B.

    def with_prefix(prefix):
        return lambda func: lambda *args, **kwargs: prefix + str(func(*args, **kwargs))

   

- C.

    def with_prefix(prefix):
        def decorator(func):
            def wrapper(*args, **kwargs):
                return func(prefix, *args, **kwargs)
            return wrapper
        return decorator

    def with_prefix(prefix):
        def decorator(func):
            def wrapper(*args, **kwargs):

- E.

    def with_prefix(prefix):
        def wrapper(func):
            def inner(*args, **kwargs):
                return prefix + str(func(*args, **kwargs))
            return inner
        return wrapper
