    def add_repr(cls):
        if "__repr__" not in cls.__dict__:
            def __repr__(self):
                return f"<{cls.__name__} instance>"
            cls.__repr__ = __repr__
        return cls

    @add_repr
    class Test:
        pass

    obj = Test()
    print(repr(obj))
