## Пример c маршрутами и моделями

    from typing import List
    from fastapi import FastAPI
    from pydantic import BaseModel

    app = FastAPI(
        title="Bookstore API",
        description="API для управления книгами в онлайн-магазине.",
        version="1.2.3",
    )

    class Book(BaseModel):
        id: int
        title: str
        author: str
        price: float

    BOOKS_DB = [
        Book(id=1, title="1984", author="George Orwell", price=9.99),
        Book(id=2, title="Brave New World", author="Aldous Huxley", price=10.99)
    ]

    @app.get("/books", response_model=List[Book], tags=["books"], summary="Список книг")
    def list_books():
        """Возвращает список всех доступных книг."""
        return BOOKS_DB

    @app.get("/books/{book_id}", response_model=Book, tags=["books"], summary="Получить книгу по ID")
    def get_book(book_id: int):
        for b in BOOKS_DB:
            if b.id == book_id:
                return b
        return {"detail": "Not Found"}  # или HTTPException(status_code=404)

### Разбор

- `title`, `description`, `version` в `FastAPI(...)` — метаданные API.

- `tags=["books"]` и `summary="..."` — дополнительно улучшают документацию.

- `response_model=List[Book]` — Swagger UI знает, что вернётся список `Book`.
