## Streamlit-дашборд на базе витрины

**Цель.** Сверстать минимально-полезный дашборд на *Streamlit*, который читает витрины из шага 1
 (`dm/daily_by_bundle.jsonl`, `dm/daily_overall.jsonl`) и показывает: ключевые KPI, тренды по дням,
 сравнение бандлов, guardrails (PASS/FAIL), а также выгрузку CSV.

### 1) Что будет уметь дашборд

- Фильтры: диапазон дат, выбор бандлов, выбор метрики CSAT (raw / weighted / smoothed).
- Карточки KPI: Requests, CSAT (с доверительным интервалом), Task Success, Helpful, p95 latency, <!-- MathML: <math xmlns="http://www.w3.org/1998/Math/MathML"><semantics><mrow><mi mathvariant="normal">/</mi><mi>r</mi><mi>e</mi><mi>q</mi><mo separator="true">,</mo></mrow><annotation encoding="application/x-tex">/req, </annotation></semantics></math> --> /req,/req,/req,/helpful.
- Тренды: линия CSAT по дням (по бандлам), столбцы Helpful Rate/TSR, p95, стоимость.
- Guardrails по выбранному дню: сравнение канарейки с базовым бандлом (PASS/FAIL) по CSAT_sm, $/helpful, p95.
- Кнопки выгрузки CSV по текущим срезам.


### 2) Установка

    pip install streamlit pandas plotly

### 3) Код дашборда

Сохраните как `quality_dashboard_step2.py` в той же папке, где лежит каталог `dm/` из шага 1.

    # quality_dashboard_step2.py
    import os
    import json
    import pandas as pd
    import plotly.express as px
    import streamlit as st
    from datetime import date

    DM_DIR = os.environ.get("DM_DIR", "dm")
    FILE_BUNDLE = os.path.join(DM_DIR, "daily_by_bundle.jsonl")
    FILE_OVER   = os.path.join(DM_DIR, "daily_overall.jsonl")

    # --- Guardrails (пороговые значения; можно править через ENV) ---
    CSAT_DROP_MAX = float(os.environ.get("CSAT_DROP_MAX", "0.02"))       # допустимое падение smoothed CSAT vs baseline (2 п.п.)
    COST_INCREASE_MAX = float(os.environ.get("COST_INCREASE_MAX", "0.10"))# допустимый рост $/helpful (+10%)
    P95_LATENCY_MAX_DELTA = float(os.environ.get("P95_LATENCY_MAX_DELTA", "0.10"))  # допустимый рост p95 (+10%)

    # --- Вспомогательные функции ---
    def pct(x):
        return None if pd.isna(x) else 100.0 * x

    def read_jsonl(path):
        if not os.path.exists(path):
            return pd.DataFrame()
        return pd.read_json(path, lines=True)

    def kpi_card(label, value, suffix="", help_text=None):
        c = st.container()
        with c:
            st.metric(label, f"{value}{suffix}" if value is not None else "—", help=help_text)

    def format_money(x):
        if pd.isna(x) or x==float("inf"):
            return "∞"
        return f"${x:.4f}"

    def guardrails_row(b, m, base):
        # d_csat = падение smoothed csat vs baseline (требуем не хуже -CSAT_DROP_MAX)
        d_csat = m["csat_sm"] - base["csat_sm"]
        # rel cost = относительный рост $/helpful
        if base["cost_per_helpful"]==float("inf"):
            rel_cost = float("inf") if m["cost_per_helpful"]!=float("inf") else 0.0
        else:
            rel_cost = (m["cost_per_helpful"]/base["cost_per_helpful"] - 1.0)
        # latency
        lat_ok = (m["p95_ms"] <= base["p95_ms"]*(1.0 + P95_LATENCY_MAX_DELTA))
        csat_ok = (d_csat >= -CSAT_DROP_MAX)
        cost_ok = (rel_cost <= COST_INCREASE_MAX)
        status = "BASE" if b==base.name else ("PASS ✅" if (csat_ok and cost_ok and lat_ok) else "FAIL ❌")
        return d_csat, rel_cost, lat_ok, csat_ok, cost_ok, status

    # --- Загрузка данных ---
    st.set_page_config(page_title="LLMOps Quality Dashboard", layout="wide")
    st.title("Quality Dashboard (LLMOps PRO)")

    df_b = read_jsonl(FILE_BUNDLE)
    df_o = read_jsonl(FILE_OVER)

    if df_b.empty or df_o.empty:
        st.error("Не найдены витрины. Сначала выполните шаг 1 и создайте dm/daily_by_bundle.jsonl и dm/daily_overall.jsonl")
        st.stop()

    # Приведение типов
    for col in ["csat_raw","csat_w","csat_sm","tsr","helpful_rate"]:
        if col in df_b.columns: df_b[col] = pd.to_numeric(df_b[col], errors="coerce")
        if col in df_o.columns: df_o[col] = pd.to_numeric(df_o[col], errors="coerce")

    if "day" in df_b.columns: df_b["day"] = pd.to_datetime(df_b["day"]).dt.date
    if "day" in df_o.columns: df_o["day"] = pd.to_datetime(df_o["day"]).dt.date

    # --- Sidebar фильтры ---
    min_day, max_day = df_o["day"].min(), df_o["day"].max()
    d_from, d_to = st.sidebar.date_input("Диапазон дат", (min_day, max_day), min_value=min_day, max_value=max_day)
    if isinstance(d_from, tuple):  # на случай старых версий streamlit
        d_from, d_to = d_from[0], d_from[1]

    metric_kind = st.sidebar.selectbox("Метрика CSAT", ["csat_sm","csat_w","csat_raw"], index=0,
                                       help="Рекомендуем использовать smoothed CSAT для принятия решений")
    bundles_all = sorted(df_b["bundle"].unique())
    sel_bundles = st.sidebar.multiselect("Бандлы", bundles_all, default=bundles_all)

    # базовый бандл — с наибольшим N в выбранном диапазоне дат
    df_range = df_b[(df_b["day"]>=d_from) & (df_b["day"]<=d_to) & (df_b["bundle"].isin(sel_bundles))]
    base_bundle = (df_range.groupby("bundle")["N"].sum().sort_values(ascending=False).index[0]
                   if not df_range.empty else bundles_all[0])

    st.sidebar.markdown("---")
    base_bundle = st.sidebar.selectbox("Базовый бандл (для guardrails)", bundles_all,
                                       index=bundles_all.index(base_bundle) if base_bundle in bundles_all else 0)

    # --- Верхние KPI (по выбранному диапазону) ---
    over = df_o[(df_o["day"]>=d_from) & (df_o["day"]<=d_to)]
    agg = {
        "N": over["N"].sum(),
        "csat": over[metric_kind].mean(),
        "tsr": over["tsr"].mean(),
        "helpful": over["helpful_rate"].mean(),
        "p95": over["p95_ms"].mean(),
        "cost_req": over["cost_per_req"].mean(),
    }
    c1,c2,c3,c4,c5,c6 = st.columns(6)
    with c1: kpi_card("Requests", int(agg["N"]))
    with c2: kpi_card("CSAT", f"{pct(agg['csat']):.1f}%" if agg["csat"]==agg["csat"] else None)
    with c3: kpi_card("Task Success", f"{pct(agg['tsr']):.1f}%")
    with c4: kpi_card("Helpful Rate", f"{pct(agg['helpful']):.1f}%")
    with c5: kpi_card("p95 latency", f"{agg['p95']:.0f}", " ms")
    with c6: kpi_card("$ per req", f"{agg['cost_req']:.4f}", "")

    st.markdown("### Тренды по дням")
    # Линия CSAT по бандлам
    ts = df_b[(df_b["day"]>=d_from) & (df_b["day"]<=d_to) & (df_b["bundle"].isin(sel_bundles))]
    if ts.empty:
        st.info("Нет данных для выбранных фильтров.")
    else:
        fig = px.line(ts.sort_values("day"), x="day", y=metric_kind, color="bundle",
                      markers=True, title=f"{metric_kind} по дням")
        fig.update_layout(yaxis_tickformat=".0%")
        st.plotly_chart(fig, use_container_width=True)

        c7, c8 = st.columns(2)
        with c7:
            fig2 = px.bar(ts, x="day", y="helpful_rate", color="bundle", barmode="group", title="Helpful Rate по дням")
            fig2.update_layout(yaxis_tickformat=".0%")
            st.plotly_chart(fig2, use_container_width=True)
        with c8:
            fig3 = px.bar(ts, x="day", y="p95_ms", color="bundle", barmode="group", title="p95 latency по дням (ms)")
            st.plotly_chart(fig3, use_container_width=True)

    st.markdown("### Guardrails (по выбранному дню)")
    # Выбор дня для guardrails (последний в диапазоне по умолчанию)
    day_for_guard = st.selectbox("День", sorted(df_range["day"].unique()) if not df_range.empty else [max_day],
                                 index=len(sorted(df_range["day"].unique()))-1 if not df_range.empty else 0)
    slice_day = df_b[(df_b["day"]==day_for_guard) & (df_b["bundle"].isin(sel_bundles))].copy()
    if slice_day.empty:
        st.info("Нет данных для выбранного дня.")
    else:
        base_row = slice_day[slice_day["bundle"]==base_bundle].iloc[0]
        slice_day = slice_day.set_index("bundle")
        rows = []
        for b, row in slice_day.iterrows():
            d_csat, rel_cost, lat_ok, csat_ok, cost_ok, status = guardrails_row(b, row, base_row)
            rows.append({
                "bundle": b,
                "N": int(row["N"]),
                "CSAT_sm": row["csat_sm"],
                "ΔCSAT_sm vs base": d_csat,
                "p95_ms": row["p95_ms"],
                "$/helpful": row["cost_per_helpful"],
                "rel $/helpful": rel_cost,
                "Guardrails": status
            })
        out = pd.DataFrame(rows).sort_values(["Guardrails","bundle"])
        # Форматирование
        fmt = {"CSAT_sm": "{:.1%}".format,
               "ΔCSAT_sm vs base": "{:+.1%}".format,
               "p95_ms": "{:.0f}".format,
               "$/helpful": format_money,
               "rel $/helpful": "{:+.0%}".format}
        st.dataframe(out.assign(**{k: out[k].map(fmt[k]) for k in fmt if k in out.columns}),
                     use_container_width=True, hide_index=True)

    st.markdown("### Срез «по бандлам» за период")
    pivot = (df_range
             .groupby("bundle", as_index=False)
             .agg(N=("N","sum"),
                  csat=(metric_kind,"mean"),
                  tsr=("tsr","mean"),
                  helpful=("helpful_rate","mean"),
                  p95=("p95_ms","mean"),
                  cost_req=("cost_per_req","mean"),
                  cost_help=("cost_per_helpful","mean")))
    if not pivot.empty:
        fmt2 = {"csat": "{:.1%}".format, "tsr": "{:.1%}".format, "helpful": "{:.1%}".format,
                "p95": "{:.0f}".format, "cost_req": "{:.4f}".format}
        shown = pivot.assign(**{k: pivot[k].map(fmt2[k]) for k in fmt2 if k in pivot.columns})
        st.dataframe(shown.rename(columns={
            "bundle":"bundle", "N":"N", "csat":"CSAT", "tsr":"TSR",
            "helpful":"Helpful", "p95":"p95 ms", "cost_req":"$/req", "cost_help":"$/helpful"
        }), use_container_width=True, hide_index=True)

    # --- Выгрузка CSV ---
    st.markdown("### Экспорт")
    colA, colB = st.columns(2)
    with colA:
        st.download_button("Скачать daily_by_bundle.csv",
                           data=df_b.to_csv(index=False).encode("utf-8"),
                           file_name="daily_by_bundle.csv", mime="text/csv")
    with colB:
        st.download_button("Скачать daily_overall.csv",
                           data=df_o.to_csv(index=False).encode("utf-8"),
                           file_name="daily_overall.csv", mime="text/csv")

### 4) Как запускать

1. Убедитесь, что после шага 1 у вас есть папка `dm/` с файлами `daily_by_bundle.jsonl` и `daily_overall.jsonl`.
2. Запустите дашборд:

    streamlit run quality_dashboard_step2.py

3. (Опционально) Настройте пороги guardrails через переменные окружения:

    export CSAT_DROP_MAX=0.03
    export COST_INCREASE_MAX=0.15
    export P95_LATENCY_MAX_DELTA=0.08
    streamlit run quality_dashboard_step2.py

### 5) Мини-лаба (что проверить)

- Выберите диапазон дат и сравните бандлы по *csat_sm*. Поменяйте «Базовый бандл» и посмотрите, как меняется статус PASS/FAIL.
- Сузьте выбор до дня с канарейкой — таблица guardrails покажет, проходит ли промо.
- Скачайте CSV и проверьте, совпадают ли агрегаты с отчётами из урока «User-oriented метрики».
