## Мини-датамарт: одна «витрина» метрик по дням и бандлам

**Цель.** Прежде чем рисовать графики, соберём *единую витрину* метрик (datamart), которую дашборд сможет читать за O(1). На выходе — два JSONL-файла: `dm/daily_by_bundle.jsonl` и `dm/daily_overall.jsonl` с метриками CSAT (raw/weighted/smoothed), TSR, Helpful, p95 latency, <!-- MathML: <math xmlns="http://www.w3.org/1998/Math/MathML"><semantics><mrow><mi mathvariant="normal">/</mi><mi>r</mi><mi>e</mi><mi>q</mi><mo separator="true">,</mo></mrow><annotation encoding="application/x-tex">/req, </annotation></semantics></math> --> /req,/req,/req,/helpful.

### 1) Что и почему агрегируем

- **По дням** (табличные строки) и **по бандлам** — чтобы строить тренды и сравнивать версии.
- **CSAT**: raw (по голосам), weighted (IPS-lite с поправкой на «склонность голосовать»), smoothed (Beta-сглаживание).
- **Прокси-метрики**: Task Success Rate (ok & без re-ask & без эскалаций), Helpful Rate, p95 latency, стоимость.
- Формат JSONL позволит легко читать данные из любого UI (Streamlit/Node/React/Vega и т.д.).


### 2) Входные события

Используем те же, что в уроке «User-oriented метрики» (шаг 1–2): `uo_events.jsonl` со строками `request.done`, `llm.tokens`, `feedback.add`, `escalate.create`.

### 3) Скрипт агрегатора (самодостаточно)

Сохраните как `quality_datamart_step1.py`. Зависимости — только стандартная библиотека.

    # quality_datamart_step1.py
    from __future__ import annotations
    import json, math, os
    from dataclasses import dataclass
    from pathlib import Path
    from collections import defaultdict
    from typing import Dict, List, Tuple

    # --- настройки IPS-lite и сглаживания (как в User-oriented/шаг 2) ---
    P_MIN = float(os.getenv("P_MIN", 0.20))     # минимальная "склонность к голосу"
    W_SILENT = float(os.getenv("W_SILENT", 0.25))  # вес "тихого успеха"
    PRIOR_ALPHA = float(os.getenv("PRIOR_ALPHA", 0.5))  # Jeffreys-like prior
    PRIOR_BETA  = float(os.getenv("PRIOR_BETA", 0.5))

    DATA = Path("uo_events.jsonl")
    OUT_DIR = Path("dm"); OUT_DIR.mkdir(exist_ok=True, parents=True)
    OUT_BUNDLE = OUT_DIR / "daily_by_bundle.jsonl"
    OUT_OVER   = OUT_DIR / "daily_overall.jsonl"
    OUT_LATEST = OUT_DIR / "latest_summary.json"

    @dataclass
    class Req:
        trace_id: str
        day: str
        user_id: str
        bundle_id: str
        status: str
        had_followup: bool
        escalated: bool
        latency_ms: int
        cost_usd: float = 0.0
        pos: int = 0
        neg: int = 0

    def wilson(p: float, n: float, z: float = 1.96) -> Tuple[float,float]:
        if n <= 0 or p!=p: return (0.0, 0.0)
        denom = 1 + z*z/n
        center = (p + z*z/(2*n))/denom
        half = (z*math.sqrt(p*(1-p)/n + z*z/(4*n*n)))/denom
        return (max(0.0, center-half), min(1.0, center+half))

    def p95(vals: List[float]) -> float:
        if not vals: return 0.0
        xs=sorted(vals); k=(len(xs)-1)*0.95
        f=int(k); c=min(f+1,len(xs)-1)
        return xs[f] if f==c else xs[f]+(xs[c]-xs[f])*(k-f)

    def load_events() -> Dict[str, Req]:
        if not DATA.exists():
            raise SystemExit("Нет uo_events.jsonl — подготовьте файл из предыдущего урока.")
        reqs: Dict[str, Req] = {}
        for line in DATA.read_text(encoding="utf-8").splitlines():
            if not line.strip(): continue
            e=json.loads(line)
            ev=e.get("event"); tid=e.get("trace_id")
            if ev=="request.done":
                reqs[tid]=Req(
                    trace_id=tid, day=e["ts"][:10], user_id=e["user_id"],
                    bundle_id=e["bundle_id"], status=e["status"],
                    had_followup=bool(e.get("had_followup", False)),
                    escalated=False, latency_ms=int(e.get("latency_ms",0))
                )
            elif ev=="llm.tokens" and tid in reqs:
                reqs[tid].cost_usd += float(e.get("cost_usd",0.0))
            elif ev=="feedback.add" and tid in reqs:
                v=int(e.get("vote",0))
                if v>0: reqs[tid].pos+=1
                elif v<0: reqs[tid].neg+=1
            elif ev=="escalate.create" and tid in reqs:
                reqs[tid].escalated=True
        return reqs

    def user_propensity(rows: List[Req]) -> Dict[str,float]:
        per=defaultdict(lambda:{"req":0,"voted":0})
        for r in rows:
            per[r.user_id]["req"] += 1
            if r.pos+r.neg>0: per[r.user_id]["voted"]+=1
        return {u:(v["voted"]/v["req"] if v["req"] else 0.0) for u,v in per.items()}

    def kish_neff(ws: List[float]) -> float:
        if not ws: return 0.0
        s=sum(ws); s2=sum(w*w for w in ws)
        return (s*s/s2) if s2>0 else 0.0

    def metrics(rows: List[Req], alpha: float, beta: float) -> Dict[str, float]:
        N=len(rows);
        pos=sum(r.pos for r in rows); neg=sum(r.neg for r in rows); votes=pos+neg
        # TSR/Helpful/latency/cost
        tsr=sum(1 for r in rows if (r.status=="ok" and not r.had_followup and not r.escalated))/N if N else 0.0
        helpful=sum(1 for r in rows if (r.pos>0 or (r.status=="ok" and not r.had_followup and not r.escalated)))
        helpful_rate=helpful/N if N else 0.0
        p95_ms=p95([r.latency_ms for r in rows])
        cost_sum=sum(r.cost_usd for r in rows)
        cost_per_req=(cost_sum/N) if N else 0.0
        cph=(cost_sum/helpful) if helpful>0 else float("inf")

        # RAW CSAT
        raw = (pos/votes) if votes>0 else float("nan")
        raw_lo, raw_hi = wilson(raw, votes) if votes>0 else (0.0,0.0)

        # WEIGHTED CSAT (IPS-lite) + "тихий успех"
        prop=user_propensity(rows)
        w_pos=[]; w_neg=[]
        for r in rows:
            p=max(prop.get(r.user_id,0.0), P_MIN)
            w=1.0/p
            if r.pos>0: w_pos.extend([w]*r.pos)
            if r.neg>0: w_neg.extend([w]*r.neg)
            if (r.pos+r.neg)==0 and r.status=="ok" and not r.had_followup and not r.escalated:
                w_pos.append(w*W_SILENT)
        sp=sum(w_pos); sn=sum(w_neg); total=sp+sn
        w_csat=(sp/total) if total>0 else float("nan")
        n_eff=kish_neff(w_pos+w_neg)
        w_lo, w_hi = wilson(w_csat, n_eff) if total>0 else (0.0,0.0)

        # SMOOTHED CSAT (на взвешенных "голосах")
        sm = ((sp+alpha)/(sp+sn+alpha+beta)) if (sp+sn+alpha+beta)>0 else float("nan")
        sm_lo, sm_hi = wilson(sm, n_eff+(alpha+beta)) if (n_eff+(alpha+beta))>0 else (0.0,0.0)

        return {
            "N":N,"votes":votes,
            "csat_raw":raw,"csat_raw_lo":raw_lo,"csat_raw_hi":raw_hi,
            "csat_w":w_csat,"csat_w_lo":w_lo,"csat_w_hi":w_hi,
            "csat_sm":sm,"csat_sm_lo":sm_lo,"csat_sm_hi":sm_hi,
            "tsr":tsr,"helpful_rate":helpful_rate,
            "p95_ms":p95_ms,"cost_per_req":cost_per_req,"cost_per_helpful":cph
        }

    def group(rows: List[Req], key_fn):
        out=defaultdict(list)
        for r in rows: out[key_fn(r)].append(r)
        return out

    def write_jsonl(path: Path, records: List[dict]):
        with path.open("w", encoding="utf-8") as f:
            for r in records:
                f.write(json.dumps(r, ensure_ascii=False)+"\n")

    def main():
        reqs = list(load_events().values())
        if not reqs:
            raise SystemExit("Нет request.done событий в uo_events.jsonl")

        # prior по всем голосам
        g_pos=sum(r.pos for r in reqs); g_neg=sum(r.neg for r in reqs)
        alpha=PRIOR_ALPHA+g_pos; beta=PRIOR_BETA+g_neg

        # по дням×бандлам
        recs_bundle=[]
        by_day = group(reqs, key_fn=lambda r: r.day)
        for day, day_rows in sorted(by_day.items()):
            by_b = group(day_rows, key_fn=lambda r: r.bundle_id)
            for b, rows in sorted(by_b.items()):
                m = metrics(rows, alpha, beta)
                recs_bundle.append({"day":day, "bundle":b, **m})

        # по дням (все бандлы)
        recs_over=[]
        for day, day_rows in sorted(by_day.items()):
            m = metrics(day_rows, alpha, beta)
            recs_over.append({"day":day, **m})

        write_jsonl(OUT_BUNDLE, recs_bundle)
        write_jsonl(OUT_OVER, recs_over)

        # summary для «последнего дня»
        latest = recs_over[-1] if recs_over else {}
        OUT_LATEST.write_text(json.dumps({
            "latest_day": latest.get("day"),
            "overview": latest
        }, ensure_ascii=False, indent=2), encoding="utf-8")

        print(f"OK → {OUT_BUNDLE}\nOK → {OUT_OVER}\nOK → {OUT_LATEST}")

    if __name__ == "__main__":
        main()

### 4) Как запустить

1. Положите рядом `uo_events.jsonl` (из урока «User-oriented метрики»).
2. Выполните:

    python quality_datamart_step1.py

Скрипт создаст `dm/daily_by_bundle.jsonl`, `dm/daily_overall.jsonl` и `dm/latest_summary.json`.

3. (Опционально) Подстройте параметры через переменные окружения:

    export P_MIN=0.3
    export W_SILENT=0.2
    export PRIOR_ALPHA=1.0
    export PRIOR_BETA=1.0
    python quality_datamart_step1.py

### 5) Мини-лаба (что проверить)

- Откройте `dm/daily_by_bundle.jsonl` и убедитесь, что у каждого дня/бандла присутствуют поля *csat_raw / csat_w / csat_sm*, *tsr*, *helpful_rate*, *p95_ms*, *$*.
- Измените `P_MIN` и `W_SILENT` и сравните, меняется ли порядок бандлов по *csat_sm* на одном и том же дне.
- Добавьте в `uo_events.jsonl` новый день — после перезапуска появится новая строка в обоих файлах JSONL.
