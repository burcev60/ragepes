    # router.py
    from __future__ import annotations

    from dataclasses import dataclass, field
    from typing import Callable, Dict, List, Optional, Tuple
    from collections import Counter, defaultdict
    import re
    import time
    import math

    # =========================
    # Типы
    # =========================

    @dataclass
    class HandlerResult:
        ok: bool
        answer: str
        route: str
        source: Optional[str] = None  # например, "kb:ustav_company p.2" или URL

    @dataclass
    class RouteEvent:
        ts: float
        query: str
        primary_route: str
        final_route: str
        ok: bool
        fallback_used: bool
        corrected_misroute: bool

    # =========================
    # Маршрутизатор (правила/LLM)
    # =========================

    class QueryRouter:
        """
        Простой маршрутизатор по правилам.
        Можно расширить LLM-классификатором, переопределив метод route().
        """
        def __init__(self) -> None:
            # базовые паттерны
            self.routes: Dict[str, List[re.Pattern]] = {
                "Track": [
                    re.compile(r"\b(заказ|трек|track|tracking|доставка|посылк)\w*\b", re.I),
                    re.compile(r"\b(order|shipment|where is my)\b", re.I),
                ],
                "Web": [
                    re.compile(r"\b(кто|когда|сколько|новост|погода|курс|президент|столица)\w*\b", re.I),
                    re.compile(r"\b(who|when|how many|news|weather|president|capital)\b", re.I),
                ],
                "RAG": [
                    re.compile(r"\b(аккаунт|настройк|инструкци|регламент|внутренн|политик)\w*\b", re.I),
                    re.compile(r"\b(reset|account|policy|internal|kb|doc)\b", re.I),
                ],
            }
            self.default_route = "RAG"

        def add_route(self, name: str, patterns: List[str]) -> None:
            """Добавить новый маршрут через список regex-паттернов."""
            compiled = [re.compile(p, re.I) for p in patterns]
            self.routes[name] = self.routes.get(name, []) + compiled

        def route(self, query: str) -> str:
            """Возвращает имя маршрута: 'RAG' | 'Web' | 'Track' | ..."""
            q = (query or "").strip()
            for name, pats in self.routes.items():
                if any(p.search(q) for p in pats):
                    return name
            return self.default_route

    # =========================
    # Отчёт/метрики маршрутов
    # =========================

    @dataclass
    class RouteReport:
        total: int = 0
        routed_counts: Counter = field(default_factory=Counter)        # по первичному выбору
        final_counts: Counter = field(default_factory=Counter)         # по финальной ветке после fallback
        successes: int = 0
        fallbacks: int = 0
        misroutes_corrected: int = 0
        events: List[RouteEvent] = field(default_factory=list)
        web_keywords: Counter = field(default_factory=Counter)         # частые слова, приведшие к Web

        def record(self, ev: RouteEvent) -> None:
            self.total += 1
            self.routed_counts[ev.primary_route] += 1
            self.final_counts[ev.final_route] += 1
            if ev.ok:
                self.successes += 1
            if ev.fallback_used:
                self.fallbacks += 1
            if ev.corrected_misroute:
                self.misroutes_corrected += 1
            self.events.append(ev)
            # собрать ключевые слова для web-маршрута
            if ev.final_route == "Web":
                for w in tokenize(ev.query):
                    if len(w) >= 3 and w not in STOPWORDS:
                        self.web_keywords[w] += 1

        def render(self) -> str:
            def pct(x: int, d: int) -> float:
                return (x / d * 100.0) if d else 0.0

            lines = []
            lines.append("—" * 5 + " Routes Report " + "—" * 5)
            lines.append(f"Total queries: {self.total}")
            for name, cnt in self.final_counts.items():
                lines.append(f"Routed to {name}: {cnt} ({pct(cnt, self.total):.0f}%)")
            lines.append(f"Successful answers: {self.successes} ({pct(self.successes, self.total):.0f}%)")
            lines.append(f"Fallback used: {self.fallbacks}")
            lines.append(f"Misroutes corrected: {self.misroutes_corrected}")
            if self.web_keywords:
                common = [k for k,_ in self.web_keywords.most_common(10)]
                lines.append(f"Common keywords causing Web route: {common}")
            lines.append("—" * 28)
            return "\n".join(lines)

    # простенький токенайзер и стоп-слова для русского/английского
    STOPWORDS = set("""
    и в во не что он на я с со как а то все она так его но да ты к у же вы за бы по только ее мне было
    если ее чем всего был будет для мы они быть из же ли или при это них до их эту тот чтоб потому
    этого какой совсем ним здесь этом один почти мой тем чтобы нее кажется более всегда много слишком
    who when how the a an is are of in to for from on with about into at by as be been being was were it
    this that those these and or but not no yes do does did done than then there their them they you your
    """.split())

    def tokenize(text: str) -> List[str]:
        return re.findall(r"[A-Za-zА-Яа-яёЁ0-9]+", text.lower())

    # =========================
    # Оркестратор с fallback-логикой
    # =========================

    class RouterOrchestrator:
        """
        Вызывает handler по выбранному маршруту. Если обработчик вернул ok=False,
        пытается задать fallback-маршруты (например, RAG -> Web).
        """
        def __init__(
            self,
            router: QueryRouter,
            handlers: Dict[str, Callable[[str], HandlerResult]],
            fallback_map: Optional[Dict[str, List[str]]] = None,
            report: Optional[RouteReport] = None
        ) -> None:
            self.router = router
            self.handlers = handlers
            self.fallback_map = fallback_map or {"RAG": ["Web"], "Web": ["RAG"], "Track": ["RAG"]}
            self.report = report or RouteReport()

        def handle(self, query: str) -> HandlerResult:
            primary = self.router.route(query)
            handler = self.handlers.get(primary)
            if handler is None:
                # не найден обработчик — шлём в Web
                primary = "Web"
                handler = self.handlers["Web"]

            res = handler(query)
            final_route = primary
            fallback_used = False
            corrected = False

            if not res.ok:
                # пробуем fallback по карте
                for alt in self.fallback_map.get(primary, []):
                    alt_handler = self.handlers.get(alt)
                    if not alt_handler:
                        continue
                    alt_res = alt_handler(query)
                    if alt_res.ok:
                        res = alt_res
                        final_route = alt
                        fallback_used = True
                        corrected = True
                        break
                else:
                    # не удалось — возвращаем исходный отказ
                    fallback_used = True

            self.report.record(RouteEvent(
                ts=time.time(),
                query=query,
                primary_route=primary,
                final_route=final_route,
                ok=res.ok,
                fallback_used=fallback_used,
                corrected_misroute=corrected
            ))
            return res

    # =========================
    # Пример обработчиков (заглушки)
    # =========================

    def rag_handler(query: str) -> HandlerResult:
        """
        Имитация RAG: отвечает, если запрос выглядит «внутренним».
        """
        if re.search(r"(аккаунт|настройк|внутренн|регламент|инструкц)", query, re.I):
            return HandlerResult(True, "Ответ из базы знаний (RAG).", "RAG", source="kb:policy#reset")
        return HandlerResult(False, "Не знаю в базе знаний.", "RAG")

    def web_handler(query: str) -> HandlerResult:
        """
        Имитация Web: отвечает на общие вопросы (кто/когда/сколько и т.п.).
        """
        if re.search(r"(кто|когда|сколько|президент|столица|курс|weather|who|when|how many)", query, re.I):
            return HandlerResult(True, "Ответ из веб-поиска (Web).", "Web", source="https://example.org")
        return HandlerResult(False, "Не найдено в веб-поиске.", "Web")

    def track_handler(query: str) -> HandlerResult:
        """
        Имитация Track: отвечает, если запрос про заказ/доставку.
        """
        if re.search(r"(заказ|трек|доставк|tracking|shipment)", query, re.I):
            return HandlerResult(True, "Заказ #A123 в пути, ETA: 3 дня.", "Track", source="track:carrier#A123")
        return HandlerResult(False, "Номер заказа не распознан.", "Track")

    # =========================
    # Демонстрация
    # =========================

    if __name__ == "__main__":
        router = QueryRouter()
        # Пример добавления нового маршрута (e.g., Calculator)
        router.add_route("Calc", [r"\b(посчита|сколько будет|calculate|calc|калькулят)\w*\b"])

        # Зарегистрируем обработчики
        handlers: Dict[str, Callable[[str], HandlerResult]] = {
            "RAG": rag_handler,
            "Web": web_handler,
            "Track": track_handler,
            # "Calc": calc_handler,  # можно добавить позже
        }

        report = RouteReport()
        orch = RouterOrchestrator(router, handlers, report=report)

        demo_queries = [
            "Что такое GPT-3?",                                # Web
            "Как сбросить настройки аккаунта?",               # RAG
            "Где мой заказ #A123? Хочу трек",                 # Track
            "Кто сейчас президент Бразилии?",                 # Web
            "Покажите регламент по доступам",                 # RAG
            "Сколько будет 12*13?",                           # (попадёт в Calc, но обработчика нет → Web через map по умолчанию)
            "Когда выйдет отчёт по кварталу?",                # Web
            "Инструкция по восстановлению пароля",            # RAG
            "Где находится столица Австралии?",               # Web
            "Мой заказ застрял, как узнать статус?",          # Track
            "Расскажите про внутреннюю политику безопасности" # RAG
        ]

        # Логи по запросам
        for q in demo_queries:
            res = orch.handle(q)
            print(f'User: "{q}"')
            # пояснение — выбранная ветка и fallback
            ev = report.events[-1]
            trail = ev.primary_route if not ev.fallback_used else f"{ev.primary_route} -> {ev.final_route}"
            print(f"Route chosen: {trail} | ok={res.ok} | answer='{res.answer}' | source={res.source}")
            print("---")

        # Итоговый отчёт
        print(report.render())
